"""Replace only four authorized large diff artifacts with exact-regeneration manifests."""
from pathlib import Path
import os,subprocess,json,hashlib,tarfile,tempfile,gzip,io,shlex
ROOT=Path.cwd();OUT=ROOT/'docs/agent-runs/phase-05';HERE=OUT/'evidence/5c-precommit'
SCOPES=['build.sbt','project','modules','tests','datasets','.scalafmt.conf','.gitignore']
SPECS=[('implementation.diff','099e303311be7a1b2e2cf5655e0c7ce9fa710474'),('5c-implementation.diff','c8342d25ba04dede26af10bedc5a76691824d1ca'),('history/5c-initial-review/implementation.diff','099e303311be7a1b2e2cf5655e0c7ce9fa710474'),('history/5c-initial-review/5c-implementation.diff','c8342d25ba04dede26af10bedc5a76691824d1ca')]
ENV=dict(os.environ,GIT_OPTIONAL_LOCKS='0')
def git(*a,cwd=ROOT):return subprocess.check_output(['git',*a],cwd=cwd,env=ENV)
def sha(b):return hashlib.sha256(b).hexdigest()
def regenerate(manifest, tree):
    args=['--git-dir='+str(ROOT/'.git'),'--work-tree='+str(tree)]
    patch=git(*args,'diff','--binary','--no-ext-diff',manifest['baseRef'],'--',*manifest['scopes'],cwd=tree)
    for name in manifest['untrackedPathsAtCapture']:
        p=subprocess.run(['git',*args,'diff','--binary','--no-ext-diff','--no-index','--','/dev/null',name],cwd=tree,env=ENV,capture_output=True)
        if p.returncode not in (0,1):raise RuntimeError(p.stderr.decode())
        patch+=p.stdout
    return patch
blobs=HERE/'historical-source-blobs';blobs.mkdir(exist_ok=True)
receipts=[]
for rel,base in SPECS:
    original=OUT/rel;data=original.read_bytes();sections=data.split(b'diff --git ')[1:]
    changed=[]
    for section in sections:
        line='diff --git '+section.split(b'\n',1)[0].decode();parts=shlex.split(line)
        assert parts[2].startswith('a/') and parts[3].startswith('b/'),line
        assert parts[2][2:]==parts[3][2:],line
        changed.append(parts[3][2:])
    headpaths=set(git('ls-tree','-r','--name-only','HEAD','--',*SCOPES).decode().splitlines())
    untracked=sorted(n for n in changed if n not in headpaths)
    with tempfile.TemporaryDirectory(prefix='scholarchain-compact-diff-') as directory:
        tree=Path(directory)
        archive=git('archive',base,'--',*SCOPES)
        with tarfile.open(fileobj=io.BytesIO(archive)) as tar:tar.extractall(tree,filter='data')
        p=subprocess.run(['git','apply','--binary','--whitespace=nowarn','-'],input=data,cwd=tree,env=ENV,capture_output=True)
        if p.returncode:raise RuntimeError(p.stderr.decode())
        files=[]
        for name in sorted(n.relative_to(tree).as_posix() for n in tree.rglob('*') if n.is_file()):
            path=tree/name;b=path.read_bytes();item={'path':name,'bytes':len(b),'sha256':sha(b),'executable':bool(path.stat().st_mode&0o111)}
            current=ROOT/name
            if current.is_file() and sha(current.read_bytes())==item['sha256']:
                item['contentSource']='targetGitRefOrVerifiedWorkingTree'
            else:
                # Historical versions are small code/config bytes, retained for exact replay.
                assert len(b)<5000000,(name,len(b))
                blob=blobs/(item['sha256']+'.gz')
                if not blob.exists():blob.write_bytes(gzip.compress(b,mtime=0))
                item['historicalBlob']=str(blob.relative_to(ROOT))
            files.append(item)
        manifest={'schema':'scholarchain.evidence-diff-manifest.v1','replacesPath':str(original.relative_to(ROOT)),'originalDiff':{'bytes':len(data),'sha256':sha(data)},'baseRef':base,'captureHead':git('rev-parse','HEAD').decode().strip(),'scopes':SCOPES,'changedPaths':changed,'untrackedPathsAtCapture':untracked,'targetFiles':files,'regeneration':{'command':'python3 docs/agent-runs/phase-05/evidence/5c-precommit/regenerate-diff.py.txt --manifest '+str(original.relative_to(ROOT))+'.manifest.json --target-ref HEAD --output /private/tmp/'+original.name,'note':'After committing, target-ref identifies the cleanup commit. Before committing, verified working-tree content is accepted. Historical source blobs restore older versions. Output must match the recorded original diff hash and byte count. Existing output is never overwritten.'}}
        replay=regenerate(manifest,tree)
        assert replay==data,{'path':rel,'wanted':sha(data),'got':sha(replay),'wantBytes':len(data),'gotBytes':len(replay)}
        mp=Path(str(original)+'.manifest.json');mp.write_text(json.dumps(manifest,indent=2)+'\n')
        receipts.append({'replaced':str(original.relative_to(ROOT)),'manifest':str(mp.relative_to(ROOT)),'originalBytes':len(data),'manifestBytes':mp.stat().st_size,'originalSha256':sha(data),'exactReplayVerified':True})
# Deletion happens only after all four independently reconstructed diffs match byte-for-byte.
for rel,_ in SPECS:(OUT/rel).unlink()
(HERE/'diff-compaction-summary.json').write_text(json.dumps({'replacements':receipts,'removedBytes':sum(e['originalBytes'] for e in receipts),'manifestBytes':sum(e['manifestBytes'] for e in receipts),'historicalBlobFiles':len(list(blobs.glob('*.gz'))),'historicalCompressedBytes':sum(p.stat().st_size for p in blobs.glob('*.gz'))},indent=2)+'\n')
print(json.dumps(receipts))
