d=json.loads(meta.read_text()); split=json.loads(splitpath.read_text());rows=d['endpointRecords']
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
p=list(range(len(rows)))
def find(a):
 while p[a]!=a:p[a]=p[p[a]];a=p[a]
 return a
def union(a,b):p[find(a)]=find(b)
seen={}
for i,r in enumerate(rows):
 for h in r['anchorHashes']:
  if h in seen:union(i,seen[h])
  else:seen[h]=i
locked={h for g in split['groups'] if split['partitions'][g['groupId']]=='held-out' for h in g['anchorHashes']}
forbidden={find(i) for i,r in enumerate(rows) if locked.intersection(r['anchorHashes'])}
authors=defaultdict(list);forbidden_authors=set()
for i,r in enumerate(rows):
 m=r['mention']

chosen=sorted(cleared,key=lambda a:(hashlib.sha256(('phase5d-demo2-seed-v1:'+a).encode()).hexdigest(),a))[:6]

No labels or scoring.','humanLabelsRead':False,'candidatePairsScored':0,'httpAttempts':0}
out=base/'frozen-development-seed-screen.json';out.write_text(json.dumps(receipt,indent=2)+'\n')
