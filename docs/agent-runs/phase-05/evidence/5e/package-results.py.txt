"""Coordinator packages observed offline evaluation, provenance and cited error evidence."""
import json, hashlib, subprocess, datetime, collections
from pathlib import Path
B=Path(__file__).resolve().parent; ROOT=Path.cwd()
load=lambda p:json.loads(Path(p).read_text())
sha=lambda p:hashlib.sha256(Path(p).read_bytes()).hexdigest()
def dump(p,x): Path(p).write_text(json.dumps(x,indent=2,ensure_ascii=False)+'\n')
def info(p):
    p=Path(p);return {'path':str(p.relative_to(ROOT)),'bytes':p.stat().st_size,'sha256':sha(p)}
Q=ROOT/'docs/agent-runs/phase-05/evidence/5c-repair/review-artifacts'

queue=load(Q/'queue.json');tasks={t['pair']['pairId']:(i+1,t) for i,t in enumerate(queue['tasks'])}
commands=load(B/'commands.json'); predictions=[c for c in commands if c['name'].startswith('prepare-')]
evaluations=[c for c in commands if c['name'].startswith('evaluate-')]
assert len(predictions)==3 and len(evaluations)==8
assert all(c['exitCode']==0 for c in predictions+evaluations)
assert max(c['finishedMs'] for c in predictions)<min(c['startedMs'] for c in evaluations)
# Retain actual source-bound decisions for queued tasks; omit unrelated population output duplication.
decisions={}
for entity in ['work','researcher']:
    for outcome in ['accepted','rejected','unresolved','conflict']:
        for line in (ROOT/'target/phase5e/disabled-a'/entity/(outcome+'.jsonl')).read_text().splitlines():
            row=json.loads(line)
            if row['pair']['pairId'] in tasks: decisions[row['pair']['pairId']]=row
assert len(decisions)==300
(B/'queued-source-decisions.jsonl').write_text(''.join(json.dumps(decisions[k],separators=(',',':'),ensure_ascii=False)+'\n' for k in sorted(decisions)))
errors=[]
for part in ['development','held-out']:
    report=load(B/('disabled-a-'+part)/'evaluation.json')
    for error in report['falsePositives']+report['falseNegatives']:
        n,t=tasks[error['pairId']];d=decisions[error['pairId']]
        reasons=d['decision']['reasonCodes']
        interpretation='Authoritative identifiers disagree; policy preserves CONFLICT, while the human label is same. Version relationships/source identifier disagreement need human policy review; do not override the held-out label or identifiers.' if d['decisionPath']=='HardConflict' else 'Scored evidence did not satisfy the unchanged acceptance rule. Inspect available feature values and missing reasons; high name similarity alone cannot establish identity.' if d['decisionPath']=='Scored' else 'Exact observed identifier acceptance disagrees with the human different label; source identifiers are observations and may contain provider attribution errors.'
        errors.append({'partition':part,'taskNumber':n,'entityType':t['pair']['entityType'],**error,'reasonCodes':reasons,'featureEvidence':d['featureEvidence'],'leftEndpoint':t['pair']['left'],'rightEndpoint':t['pair']['right'],'analysis':interpretation,'queueLocator':str(Q/'queue.json'),'analysisNotIndependentIdentityAdjudication':True})
dump(B/'representative-errors.json',errors)
rows=[]
for name in ['disabled-a','fake-a']:
    for part in ['development','held-out']:
        for subset in ['', '-without-assisted']:
            r=load(B/(name+'-'+part+subset)/'evaluation.json')
            for entity,m in [('overall',r['overall']),*r['byEntity'].items()]:
                rows.append({'run':name,'partition':part,'cohort':'without-22-assisted' if subset else 'all-initial','entity':entity,**{k:m[k] for k in ['tasks','binaryGold','uncertain','tp','fp','fn','tn','precision','recall','f1','decisiveCoverage','unresolvedRate','conflictRate','missingPredictions','invalidPredictions']}})
dump(B/'metrics-summary.json',rows)
