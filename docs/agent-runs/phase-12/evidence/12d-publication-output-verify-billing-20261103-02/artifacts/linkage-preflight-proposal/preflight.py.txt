class Refused(Exception):
    pass

def need(value, code):
    if not value:
        raise Refused(code)

class PublicParser(argparse.ArgumentParser):
    def error(self, message):
        raise Refused('CLI_ARGUMENTS')

def validate_filename(value):
    need(type(value) is str and re.fullmatch(r'approval-[A-Za-z0-9_-]{1,96}\.json', value) is not None,
         'ORIGINAL_GATE_FILENAME')
    return value

def public_bytes(path):
    # Pinned public source only, no credential/config/private path.
    for parent in (path.parent, *path.parent.parents):
        need(not stat.S_ISLNK(parent.lstat().st_mode), 'PUBLIC_SOURCE_SYMLINK')
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW)
    try:
        before=os.fstat(fd)
        need(stat.S_ISREG(before.st_mode) and before.st_nlink == 1 and before.st_size <= 1024*1024,
             'PUBLIC_SOURCE_TYPE_SIZE')
        data=bytearray()
        while True:
            chunk=os.read(fd,65536)
            if not chunk: break
            data.extend(chunk);need(len(data)<=1024*1024,'PUBLIC_SOURCE_TYPE_SIZE')
        def identity(st):return (st.st_dev,st.st_ino,st.st_size,st.st_mtime_ns,st.st_ctime_ns)
        need(identity(before)==identity(os.fstat(fd))==identity(path.lstat()),'PUBLIC_SOURCE_CHANGED')
        return bytes(data)
    finally:
        os.close(fd)

def source_check():
    env={'HOME':'/Users/danny','PATH':'/usr/bin:/bin'}
    def git(*args):
        result=subprocess.run(['/usr/bin/git','-C',str(REPO),*args],env=env,
                              capture_output=True,stdin=subprocess.DEVNULL,timeout=10)
        need(result.returncode==0,'PUBLIC_GIT_CHECK')
        return result.stdout
    need(git('rev-parse','HEAD').decode().strip()==SOURCE_COMMIT,'PUBLIC_SOURCE_COMMIT')
    need(git('status','--porcelain=v1','-z','--untracked-files=all')==b'','PUBLIC_CHECKOUT_DIRTY')
    actual={str(p.relative_to(REPO)) for p in (REPO/'scripts/billing').rglob('*') if p.is_file()}
    need(actual==set(PUBLIC_FILES),'PUBLIC_SOURCE_SET')
    for rel,expected in PUBLIC_FILES.items():
        git('ls-files','--error-unmatch','--',rel)
        need(hashlib.sha256(public_bytes(REPO/rel)).hexdigest()==expected,'PUBLIC_SOURCE_HASH')

def load_verifier():
    spec=importlib.util.spec_from_file_location('approved_local_preflight_verifier',REPO/'scripts/billing/budget_verification.py')
    module=importlib.util.module_from_spec(spec);sys.modules[spec.name]=module;spec.loader.exec_module(module)
    return module

def check_linkage(filename, verifier, policy, now):
    """Python injection is for synthetic tests only; main supplies fixed production policy."""
    filename=validate_filename(filename)
    lock=policy.primary/'.billing-wrapper-active.lock'
    need(not os.path.lexists(lock),'WRAPPER_ACTIVE')
    original_receipt=policy.primary/'operations'/verifier.ORIGINAL_OPERATION/'apply.receipt.json'
    raw=verifier.read_private(original_receipt,policy.uid)
    raw_sha=hashlib.sha256(raw).hexdigest()
    # These are narrow in-memory LINKAGE ARGUMENTS, not an execution gate or approval.
    arguments={'original_gate_filename':filename,'original_apply_receipt_sha256':raw_sha,
               'aws_cli_sha256':AWS_LAUNCHER_SHA,'window_start':now.isoformat()}
    ex=verifier.Verification(policy)
    ex.gate=arguments
    linked=ex.linkage()  # Only the accepted read-only metadata/opaque-hash method, no stage.
    need(linked=={'original_operation':verifier.ORIGINAL_OPERATION,
                  'original_gate_sha256':verifier.ORIGINAL_GATE_SHA,
                  'original_apply_receipt_sha256':raw_sha,'state_backup_verified':True},'LINKAGE_RESULT')
    need(not os.path.lexists(lock),'WRAPPER_ACTIVE')
    return {'status':'LOCAL_LINKAGE_VALIDATED','source_commit':SOURCE_COMMIT,
            'original_operation':verifier.ORIGINAL_OPERATION,
            'original_gate_sha256':verifier.ORIGINAL_GATE_SHA,
            'original_apply_receipt_sha256':raw_sha,
            'original_gate_context_match':True,'state_backup_match':True,'aws_cli_launcher_hash_match':True,
            'active_wrapper_absent_at_checks':True,'AWS_calls':0,'private_recipient_read':False,
            'execution_authorized':False,'gate_created':False,'files_modified':False}

def main(argv=None):
    try:
        parser=PublicParser(description=__doc__)
        parser.add_argument('--original-gate-filename',required=True)
        args=parser.parse_args(argv)
        validate_filename(args.original_gate_filename)
        source_check()  # Must finish before importing a policy or touching private storage.
        verifier=load_verifier()
        policy=verifier.production_policy()
        result=check_linkage(args.original_gate_filename,verifier,policy,datetime.now(timezone.utc))
        source_check()  # Do not publish a newly measured hash if public release drifted.
        print(json.dumps(result,sort_keys=True))

        return 2

        print(json.dumps({'status':'BLOCKED-ON','code':'LINKAGE_PREFLIGHT_FAILED'},sort_keys=True))
        return 2
